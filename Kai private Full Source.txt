                                local pet = petsToOffer[i]

                                if pet.Name == selectedPet then

                                    offered = offered + 1
                                    task.wait(0.1) 
                                end
                            end
                        end
                    end

                    if running then
                        tradingEvent:FireServer("acceptTrade")
                    end
                end
            end
            task.wait(1) 
        end
    end)
